cmake_minimum_required(VERSION 3.16)
project(vibeswipe_cpp VERSION 0.1.0 LANGUAGES CXX)

add_library(vibeswipe src/corpus.cpp src/scoring.cpp)
target_include_directories(vibeswipe PUBLIC include)
target_compile_features(vibeswipe PUBLIC cxx_std_17)
set_target_properties(vibeswipe PROPERTIES CXX_EXTENSIONS OFF)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(vibeswipe PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
endif()

add_executable(vibeswipe-cpp src/main.cpp)
target_link_libraries(vibeswipe-cpp PRIVATE vibeswipe)
add_executable(vibeswipe-rank src/rank_main.cpp)
target_link_libraries(vibeswipe-rank PRIVATE vibeswipe)
add_executable(corpus_benchmark bench/corpus_benchmark.cpp)
target_link_libraries(corpus_benchmark PRIVATE vibeswipe)
enable_testing()
foreach(component corpus scoring)
    add_executable(${component}_test tests/${component}_test.cpp)
    target_link_libraries(${component}_test PRIVATE vibeswipe)
    add_test(NAME ${component} COMMAND ${component}_test)
endforeach()
